% \normalsize
\begin{itemize}
    \itemsep-0.3em
    \item Maximum/Minimum flow with lower bound / Circulation problem
    %\vspace{-1em}
    \begin{enumerate}
        \itemsep-0.3em
        \item Construct super source $S$, sink $T$.
        \item Edge $(x, y, l, u)$: connect $x \rightarrow y$ with cap $u - l$.
        \item $in(v)$ := sum of incoming lower bounds minus sum of outgoing ones.
        \item $in(v) > 0$: connect $S \rightarrow v$ with cap $in(v)$; else $v \rightarrow T$ with cap $-in(v)$.
        \begin{itemize}
            \itemsep-0.2em
            \item Max: connect $t \rightarrow s$ with cap $\infty$ (skip for circulation), $f$ := maxflow $S \to T$. No solution if $f \neq \sum_{in(v) > 0}{in(v)}$; else maxflow $s \to t$ is the answer.
            \item Min: $f$ := maxflow $S \to T$; connect $t \rightarrow s$ with cap $\infty$, $f^\prime$ := flow $S \to T$. No solution if $f + f^\prime \neq \sum_{in(v) > 0}{in(v)}$; else $f^\prime$.
        \end{itemize}
        \item Each edge $e$ has solution $l_e + f_e$, $f_e$ its flow.
    \end{enumerate}
    \item Construct minimum vertex cover from maximum matching $M$ on bipartite graph $(X, Y)$
    %\vspace{-1em}
    \begin{enumerate}
        \itemsep-0.3em
        \item Redirect: $y \rightarrow x$ if $(x, y) \in M$, else $x \rightarrow y$.
        \item DFS from unmatched vertices in $X$.
        \item Choose $x \in X$ iff unvisited; $y \in Y$ iff visited.
    \end{enumerate}
    \item Minimum cost cyclic flow
    %\vspace{-0.5em}
    \begin{enumerate}
        \itemsep-0.3em
        \item Construct super source $S$, sink $T$.
        \item Edge $(x, y, c)$: if $c > 0$ connect $x \rightarrow y$ with $(cost, cap) = (c, 1)$, else $y \rightarrow x$ with $(-c, 1)$.
        \item For $c < 0$ edges, sum cost as $K$, then $d(y){+}{+}$, $d(x){-}{-}$.
        \item $d(v) > 0$: connect $S \rightarrow v$ with $(0, d(v))$; $d(v) < 0$: $v \rightarrow T$ with $(0, -d(v))$.
        \item Flow $S \to T$; answer is the flow cost $C + K$.
    \end{enumerate}
    \item Maximum density induced subgraph
    %\vspace{-1em}
    \begin{enumerate}
        \itemsep-0.3em
        \item Binary search the answer $T$; let $K$ be the sum of all weights.
        \item Connect $s \rightarrow v$, $v \in G$, with cap $K$.
        \item Edge $(u, v, w)$: connect both ways with cap $w$.
        \item Connect $v \rightarrow t$ with cap $K + 2T - (\sum_{e \in E(v)}{w(e)}) - 2w(v)$.
        \item $T$ is valid if maxflow $f < K \lvert V \rvert$.
    \end{enumerate}
    \item Minimum weight edge cover
    %\vspace{-1em}
    \begin{enumerate}
        \itemsep-0.3em
      \item Copy each $v \in V$ as $v^\prime$; connect $u^\prime \to v^\prime$ with weight $w(u, v)$.
      \item Connect $v \to v^\prime$ with weight $2\mu(v)$, $\mu(v)$ := cheapest edge at $v$.
      \item Min weight perfect matching on $G^\prime$.
    \end{enumerate}
    \item Project selection problem
    %\vspace{-1em}
    \begin{enumerate}
      \itemsep-0.3em
      \item $p_v > 0$: edge $(s, v)$ cap $p_v$; else $(v, t)$ cap $-p_v$.
      \item Edge $(u, v)$ cap $w$ := cost of choosing $u$ without $v$.
      \item The mincut equals the max profit over subsets of projects.
    \end{enumerate}
    \item Dual of minimum cost maximum flow
    %\vspace{-1em}
    \begin{enumerate}
      \itemsep-0.3em
      \item Cap $c_{uv}$, flow $f_{uv}$, cost $w_{uv}$, required flow difference $b_u$.
      \item If all $w_{uv}$ are integers, an optimum with all $p_u$ integral exists.
    \end{enumerate}
    $$
    \begin{aligned}\min\sum_{uv} w_{uv}f_{uv} \\ -f_{uv} \geq -c_{uv} \\ \sum_{v} f_{vu} - \sum_{v} f_{uv} = -b_u\end{aligned}
    \Leftrightarrow
    \begin{aligned}\min\sum_{u} b_up_u + \sum_{uv}c_{uv}\max(0, p_v - p_u - w_{uv}) \\ p_u \geq 0 \end{aligned}
    $$

\end{itemize}
