\MobiusHeading{Notation}

$[P]\in\{0,1\}$ is the indicator of $P$; $p$ is prime; $H=\min(N,M)$;
$(f*g)(n)=\sum_{d\mid n}f(d)g(n/d)$ (commutative, associative).
$\varepsilon=[n=1]$, $1(n)=1$, $\mathrm{id}_k(n)=n^k$, $\mathrm{id}=\mathrm{id}_1$,
$\sigma_k=\mathrm{id}_k*1$, $\tau=\sigma_0$, $\sigma=\sigma_1$,
$\omega(n)/\Omega(n)$ = \#distinct/\#all prime factors, $\lambda(n)=(-1)^{\Omega(n)}$.
Quotient blocks over $\lfloor N/d\rfloor,\lfloor M/d\rfloor$: $O(\sqrt N+\sqrt M)$.
Linear sieve for $\mu,\varphi$ or any multiplicative function up to $L$: $O(L)$.\par

\MobiusHeading{M\"obius inversion}

\MobiusStep{$\mu(1)=1$, $\mu(p)=-1$, $\mu(p^e)=0\ (e\ge2)$, multiplicative.}

\MobiusStep{$f=g*1\Leftrightarrow g=f*\mu$:\quad
$f(n)=\sum_{d\mid n}g(d)\Leftrightarrow g(n)=\sum_{d\mid n}\mu(d)f(n/d)$.}

\MobiusStep{$F(n)=\sum_{n\mid d}f(d)\Leftrightarrow f(n)=\sum_{n\mid d}\mu(d/n)F(d)$
\quad(finite support).}

\MobiusHeading{Dirichlet convolution identities}

\MobiusStep{$\mu*1=\varepsilon$,\quad $\varphi*1=\mathrm{id}$,\quad $\mu*\mathrm{id}=\varphi=n\prod_{p\mid n}(1-1/p)$.}

\MobiusStep{$1*1=\tau$,\quad $\mu*\tau=1$,\quad $\mu*\sigma_k=\mathrm{id}_k$,\quad
$\varphi*\tau=\sigma$,\quad $\mu*\log=\Lambda$ (von Mangoldt).}

\MobiusStep{$J_k=\mu*\mathrm{id}_k=n^k\prod_{p\mid n}(1-p^{-k})$, $J_1=\varphi$, $J_k*1=\mathrm{id}_k$.}

\MobiusStep{$\mu^2*1=2^{\omega}$,\quad $2^{\omega}*1=\tau(n^2)$,\quad
$\mu^2(n)=\sum_{d^2\mid n}\mu(d)$.}

\MobiusStep{$\lambda*1=[n\text{ is a square}]$,\quad $\lambda*\mu^2=\varepsilon$.}

\MobiusStep{$g$ completely multiplicative: $(fg)*(hg)=(f*h)g$, $g^{-1}=\mu g$.
$\rightarrow(\mu\,\mathrm{id}_k)*\mathrm{id}_k=\varepsilon$,
$(\varphi\,\mathrm{id})*\mathrm{id}=\mathrm{id}_2$,
$(\mu\,\mathrm{id})*\mathrm{id}=\varepsilon$.}

\MobiusStep{$f$ multiplicative: $f*g$ and $f^{-1}$ multiplicative, $f^{-1}(p)=-f(p)$;
$\sum_{d\mid n}\mu(d)f(d)=\prod_{p\mid n}(1-f(p))$.}

\MobiusStep{$\tau(p^e)=e+1$, $\varphi(p^e)=p^{e-1}(p-1)$,
$\sigma_k(p^e)=(p^{k(e+1)}-1)/(p^k-1)$.}

\MobiusStep{$[\gcd(i,j)=1]=\sum_{d\mid\gcd}\mu(d)$,\quad
$\gcd(i,j)=\sum_{d\mid\gcd}\varphi(d)$,\quad
$h(\gcd(i,j))=\sum_{d\mid\gcd}(\mu*h)(d)$.}

\MobiusStep{$\tau(ij)=\sum_{x\mid i}\sum_{y\mid j}[\gcd(x,y)=1]$,\quad
$\sigma(ij)=\sum_{x\mid i}\sum_{y\mid j}[\gcd(x,y)=1]\,xj/y$.}

\MobiusStep{$\varphi(ij)=\varphi(i)\varphi(j)\,g/\varphi(g)$, $g=\gcd(i,j)$.}

\MobiusStep{$\sum_{i\le n,\gcd(i,n)=1}i=(n\varphi(n)+[n=1])/2$,\quad
$\sum_{i=1}^n\mathrm{lcm}(i,n)=\frac n2\bigl(1+\sum_{d\mid n}d\varphi(d)\bigr)$.}

\MobiusStep{Any $c=f*g$ up to $L$ by looping multiples: $O(L\log L)$.}

\MobiusHeading{Results ($\sum_{i\le N}\sum_{j\le M}$)}

\MobiusStep{$[\gcd(i,j)=1]\rightarrow\sum_{d=1}^H\mu(d)\lfloor N/d\rfloor\lfloor M/d\rfloor$;\quad
$[\gcd=g]$: same with $N,M\rightarrow\lfloor N/g\rfloor,\lfloor M/g\rfloor$.}

\MobiusStep{$h(\gcd(i,j))\rightarrow\sum_{d=1}^H(\mu*h)(d)\lfloor N/d\rfloor\lfloor M/d\rfloor$;\quad
$\gcd\rightarrow c=\varphi$,\quad $\gcd^s\rightarrow c=J_s$.}

\MobiusStep{$\mathrm{lcm}(i,j)\rightarrow\sum_{T=1}^H S(\lfloor N/T\rfloor)S(\lfloor M/T\rfloor)\,c(T)$,
$S(x)=\frac{x(x+1)}2$, $c(T)=T\prod_{p\mid T}(1-p)$.}

\MobiusStep{$\tau(ij)\rightarrow\sum_{d=1}^H\mu(d)F(\lfloor N/d\rfloor)F(\lfloor M/d\rfloor)$,
$F(x)=\sum_{i\le x}\lfloor x/i\rfloor$.}

\MobiusStep{$k$ variables:
$\sum_{i_1..i_k\le N}[\gcd=1]=\sum_d\mu(d)\lfloor N/d\rfloor^k$,\quad
$\sum_{i_1..i_k\le N}\gcd=\sum_T\varphi(T)\lfloor N/T\rfloor^k$.}

\MobiusStep{$\sum_{i=1}^k\gcd(i,n)=\sum_{d\mid n}\varphi(d)\lfloor k/d\rfloor$,\quad
$\sum_{a\le x}[\gcd(a,m)=1]=\sum_{e\mid m}\mu(e)\lfloor x/e\rfloor$.}

\MobiusHeading{Prefix sums (Du Jiao)}

\MobiusStep{$f*g=h$, $S_f(N)=\sum_{i\le N}f(i)$:
$g(1)S_f(N)=S_h(N)-\sum_{i=2}^N g(i)S_f(\lfloor N/i\rfloor)$.
Sieve up to $N^{2/3}$ + memo: $O(N^{2/3})$.}

\MobiusStep{$\mu$: $g=1,h=\varepsilon$;\quad $\varphi$: $g=1,h=\mathrm{id}$;\quad
$\mu\,\mathrm{id}$: $g=\mathrm{id},h=\varepsilon$;\quad $\varphi\,\mathrm{id}$: $g=\mathrm{id},h=\mathrm{id}_2$.}

\MobiusHeading{Dirichlet prefix sum, $O(L\log\log L)$}

\MobiusStep{For each prime $p$:
$f*1$: \texttt{for i=1..L/p: f[i*p]+=f[i]};
$f*\mu$: \texttt{for i=L/p..1: f[i*p]-=f[i]};
$\sum_{n\mid d}f(d)$: \texttt{for i=L/p..1: f[i]+=f[i*p]}.}

\MobiusHeading{Quotient blocks}

\MobiusStep{$q_N=\lfloor N/l\rfloor,\ q_M=\lfloor M/l\rfloor,\
r=\min(\lfloor N/q_N\rfloor,\lfloor M/q_M\rfloor)$; floors constant on $[l,r]$.}

\MobiusStep{$\rightarrow\sum_{d=l}^r c(d)\lfloor N/d\rfloor\lfloor M/d\rfloor
=q_Nq_M(P_c(r)-P_c(l-1))$; $l\leftarrow r+1$ while $l\le H$.}

\MobiusHeading{Other theorems}

\MobiusStep{\textbf{Lucas:} prime $p$,
$n=\sum_i n_i p^i$, $k=\sum_i k_i p^i$
$\rightarrow\binom nk\equiv\prod_i\binom{n_i}{k_i}\pmod p$.
Factorial/inverse-factorial preprocessing: $O(p)$ time/space;
query: $O(1+\log_p(n+1))$.}

\textbf{CRT:} compatible residues determine one solution modulo the lcm of the moduli.
For $r$ congruences and intermediate values bounded by $U$:
$O(r\log U)$ arithmetic operations.\par

\MobiusStep{\textbf{Kirchhoff:} for an undirected graph,
$L_{ii}=\deg(i)$, $L_{ij}=-\#\{\text{edges between }i,j\}$ for $i\ne j$.
Delete row and column $r$ to obtain $L^{(r)}$
$\rightarrow\#\{\text{spanning trees}\}=\det L^{(r)}$.
For $v$ vertices, Gaussian elimination: $O(v^3)$ time, $O(v^2)$ space.}

\MobiusStep{\textbf{Burnside:} $|X/G|=|G|^{-1}\sum_{g\in G}|\operatorname{Fix}(g)|$.
Cost $T_{\rm fix}$ per fixed-point count
$\rightarrow O(|G|T_{\rm fix})$ total time.}

\MobiusStep{\textbf{Pick:} simple lattice polygon, $A=I+B/2-1$ ($I$ interior, $B$ boundary points).}
